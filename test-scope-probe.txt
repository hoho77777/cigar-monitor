name: Deploy landing to Pages

on:
  push:
    branches: [main]
    paths:
      - "landing/**"
      - ".github/workflows/pages.yml"
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4
      - name: Prepare site
        run: |
          mkdir -p site
          cp -r landing/. site/
          if [ ! -f site/index.html ]; then cp "landing/产品介绍.html" site/index.html; fi
          ls -la site
      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: site
      - id: deployment
        uses: actions/deploy-pages@v4
